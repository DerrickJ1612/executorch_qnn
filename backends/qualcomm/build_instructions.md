1. sudo swapon /swapfile (if you already have a swapfile to add swap memory)

2. python install_executorch.py –clean

3. export EXECUTORCH_ROOT=$HOME/Desktop/git/qnn/executorch/

4. export QNN_SDK_ROOT=$HOME/Desktop/qnn_sdk/2.39.0.250926/

5. export LD_LIBRARY_PATH=$QNN_SDK_ROOT/lib/aarch64-ubuntu-gcc9.4/:$LD_LIBRARY_PATH

6. export PYTHONPATH=$EXECUTORCH_ROOT/..

7. export ADSP_LIBRARY_PATH=$QNN_SDK_ROOT/lib/hexagon-v68/unsigned/

8. CMAKE_ARGS="-DEXECUTORCH_BUILD_XNNPACK=OFF -DEXECUTORCH_BUILD_QNN=ON -DQNN_SDK_ROOT=$QNN_SDK_ROOT -DPYTHON_EXECUTABLE=$(which python)" ./install_executorch.sh

9. cmake -S . -B cmake-out -G Ninja -DCMAKE_INSTALL_PREFIX=$EXECUTORCH_ROOT/cmake-out/et-install -DCMAKE_BUILD_TYPE=Release -DEXECUTORCH_BUILD_QNN=ON -DEXECUTORCH_BUILD_DEVTOOLS=ON -DEXECUTORCH_BUILD_EXTENSION_LLM=ON -DEXECUTORCH_BUILD_EXTENSION_LLM_RUNNER=ON -DEXECUTORCH_BUILD_EXTENSION_MODULE=ON -DEXECUTORCH_BUILD_EXTENSION_DATA_LOADER=ON -DEXECUTORCH_BUILD_EXTENSION_FLAT_TENSOR=ON -DEXECUTORCH_BUILD_EXTENSION_TENSOR=ON -DEXECUTORCH_ENABLE_EVENT_TRACER=ON -DEXECUTORCH_ENABLE_LOGGING=ON -DQNN_SDK_ROOT=$QNN_SDK_ROOT -DEXECUTORCH_BUILD_KERNELS_QUANTIZED=ON -DPYTHON_EXECUTABLE=$PYTHON_EXECUTABLE

10. cmake --build cmake-out/ -j4

11. cmake --install cmake-out --config Release

***** END OF STAGE 1 *****

***** BUILDING QUALCOMM SPECIFIC EXAMPLES ****

12. cmake -S $EXECUTORCH_ROOT/examples/qualcomm/ -B $EXECUTORCH_ROOT/cmake-out/examples/qualcomm -DCMAKE_BUILD_TYPE=Release -DCMAKE_PREFIX_PATH="$EXECUTORCH_ROOT/cmake-out/et-install;$EXECUTORCH_ROOT/cmake-out;$EXECUTORCH_ROOT/cmake-out/third-party/gflags;" -DSUPPORT_REGEX_LOOKAHEAD=ON -DBUILD_TESTING=OFF -DEXECUTORCH_ENABLE_LOGGING=ON -DEXECUTORCH_BUILD_KERNELS_QUANTIZED=ON -DCMAKE_FIND_ROOT_PATH_MODE_PACKAGE=BOTH -DPYTHON_EXECUTABLE=$(which python)

13. make --build cmake-out/examples/qualcomm/ -j2

***** BUILDING EXECUTORCH CORE LLAMA EXAMPLES *****

14. cmake -S examples/models/llama -B cmake-out/examples/models/llama -G Ninja -DCMAKE_BUILD_TYPE=Release -DCBUILD_TESTING=OFF -DCMAKE_PREFIX_PATH="$EXECUTORCH_ROOT/cmake-out/et-install;$EXECUTORCH_ROOT/cmake-out/third-party/gflags;" -DEXECUTORCH_ENABLE_LOGGING=ON -DCMAKE_FIND_ROOT_PATH_MODE_PACKAGE=BOTH -DPYTHON_EXECUTABLE=$(which python)

15. cmake --build cmake-out/examples/models/llama/ -j4


****** BUILDING AOT COMPONENTS *******

16. First comment out the x86_64 check in /backends/qualcomm/CMakeLists.txt

17. cmake -S . -B build-aot -G Ninja   -DCMAKE_INSTALL_PREFIX="$EXECUTORCH_ROOT/build-aot/et_install"   -DEXECUTORCH_BUILD_QNN=ON   -DQNN_SDK_ROOT="$QNN_SDK_ROOT"   -DEXECUTORCH_BUILD_DEVTOOLS=ON   -DEXECUTORCH_BUILD_EXTENSION_LLM=ON   -DEXECUTORCH_BUILD_EXTENSION_LLM_RUNNER=ON   -DEXECUTORCH_BUILD_EXTENSION_MODULE=ON   -DEXECUTORCH_BUILD_EXTENSION_TENSOR=ON   -DEXECUTORCH_BUILD_EXTENSION_DATA_LOADER=ON   -DEXECUTORCH_BUILD_EXTENSION_FLAT_TENSOR=ON   -DEXECUTORCH_ENABLE_EVENT_TRACER=ON   -DEXECUTORCH_ENABLE_LOGGING=ON   -DEXECUTORCH_BUILD_PYBIND=ON   -DPYTHON_EXECUTABLE="$(which python3)"

18. cmake --build build-aot --target PyQnnManagerAdaptor PyQnnWrapperAdaptor -j4
